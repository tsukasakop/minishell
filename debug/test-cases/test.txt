- ビルドイン echo
echo

echo HELLO

echo HELLO WORLD

echo a    a

echo -n

echo -n HELLO

echo -n -n -n HELLO WORLD

echo -n HELLO -n WORLD -n

-- 以下3つ、出力がbashと異なる
echo -nnn

echo -nnn HELLO

echo -nnn  -n HELLO

- ビルドイン env
env

-- 以下2つ、出力がbashと異なる
env aaaaaa
env aaaaaa bbbbb cccc

- ビルドイン export
export
export VAR=val
echo $VAR
export STR="Hello      World"
echo $STR
export NUM=1234567
echo $NUM
export Empty=
echo $Empty
export FT42=Hello
echo $FT42
export test1=hello test2=world test3=42tokyo
echo $test1 $test2 $test3
export $aaa
echo $aaa
export VAR=test
echo $VAR
export AAA
echo $AAA

-- 以下6つ、出力がbashと異なる
export 42=val
    export | grep 42=
export 42FT=val
    export | grep 42FT=
export FT%=val
    export | grep FT%=
export $
    export | grep $=
export "   "
export num=1234567 str=hello c=a 42=tokyo test=yes
    export | grep num=
    export | grep str=
    export | grep c=
    export | grep 42=
    export | grep test=
